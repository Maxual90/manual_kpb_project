-- =====================================================
-- College Alumni Connect - database setup
-- Usage:  mysql -u root < database.sql
--   or phpMyAdmin > Import > choose this file
-- Re-running this file RESETS the 4 tables below.
-- =====================================================

CREATE DATABASE IF NOT EXISTS firstdb
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE firstdb;

DROP TABLE IF EXISTS messages;
DROP TABLE IF EXISTS posts;
DROP TABLE IF EXISTS connections;
DROP TABLE IF EXISTS students;


-- =====================================================
-- TABLES
-- =====================================================

CREATE TABLE students (
    admin_no INT          NOT NULL,
    name     VARCHAR(100) NOT NULL,
    email    VARCHAR(100) NOT NULL,
    batch    VARCHAR(20)  NOT NULL,
    course   VARCHAR(100) NOT NULL,
    pass     VARCHAR(255) NOT NULL,
    status   ENUM('pending','approved','rejected') NOT NULL DEFAULT 'pending',
    PRIMARY KEY (admin_no),
    UNIQUE KEY uq_students_email (email),
    KEY idx_students_status (status),
    KEY idx_students_batch (batch)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE connections (
    id                INT NOT NULL AUTO_INCREMENT,
    sender_admin_no   INT NOT NULL,
    receiver_admin_no INT NOT NULL,
    status            ENUM('pending','accepted','rejected') NOT NULL DEFAULT 'pending',
    created_at        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_connection_pair (sender_admin_no, receiver_admin_no),
    KEY idx_connections_receiver (receiver_admin_no),
    CONSTRAINT fk_conn_sender   FOREIGN KEY (sender_admin_no)
        REFERENCES students (admin_no) ON DELETE CASCADE,
    CONSTRAINT fk_conn_receiver FOREIGN KEY (receiver_admin_no)
        REFERENCES students (admin_no) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE posts (
    id         INT NOT NULL AUTO_INCREMENT,
    admin_no   INT NOT NULL,
    content    TEXT NULL,
    image_path VARCHAR(255) NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_posts_author (admin_no),
    KEY idx_posts_created (created_at),
    CONSTRAINT fk_posts_student FOREIGN KEY (admin_no)
        REFERENCES students (admin_no) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE messages (
    id                INT NOT NULL AUTO_INCREMENT,
    sender_admin_no   INT NOT NULL,
    receiver_admin_no INT NOT NULL,
    message           TEXT NOT NULL,
    created_at        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_messages_pair (sender_admin_no, receiver_admin_no, created_at),
    CONSTRAINT fk_msg_sender   FOREIGN KEY (sender_admin_no)
        REFERENCES students (admin_no) ON DELETE CASCADE,
    CONSTRAINT fk_msg_receiver FOREIGN KEY (receiver_admin_no)
        REFERENCES students (admin_no) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
